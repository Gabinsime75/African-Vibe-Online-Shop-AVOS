Confirm the active AWS identity

	  --query '{
		Account:Account,
		Arn:Arn,
		UserId:UserId
	  }' \
	  --output table

verifies which credentials the AWS CLI is using.
	Set session variables:
		export AVOS_ACCOUNT_ID="$(
		  aws sts get-caller-identity \
			--query Account \
			--output text
		)"

		export AVOS_REGION="us-east-2"

		printf 'AWS account: %s\n' "$AVOS_ACCOUNT_ID"
		printf 'AWS Region:  %s\n' "$AVOS_REGION"

Determine the AWS Organizations state
	aws organizations describe-organization \
	  --query 'Organization.{
		OrganizationId:Id,
		ManagementAccountId:ManagementAccountId,
		FeatureSet:FeatureSet,
		AvailablePolicyTypes:AvailablePolicyTypes
	  }' \
	  --output table
	Possible outcomes:
		- A table means the account belongs to an AWS Organization.
		- AWSOrganizationsNotInUseException means it is standalone.
		- AccessDeniedException can mean the current identity lacks Organizations permissions.

Determine whether this is the management account
	export MANAGEMENT_ACCOUNT_ID="$(
	  aws organizations describe-organization \
		--query 'Organization.ManagementAccountId' \
		--output text
	)"

	if [ "$AVOS_ACCOUNT_ID" = "$MANAGEMENT_ACCOUNT_ID" ]; then
	  echo "RESULT: this is the Organizations management account"
	else
	  echo "RESULT: this is an Organizations member account"
	fi

management-account check
	aws organizations describe-organization \
	  --query 'Organization.{
		OrganizationId:Id,
		ManagementAccountId:MasterAccountId,
		FeatureSet:FeatureSet
	  }' \
	  --output table

	export MANAGEMENT_ACCOUNT_ID="$(
	aws organizations describe-organization \
		--query 'Organization.MasterAccountId' \
		--output text
	)"

	printf 'Current account:    %s\n' "$AVOS_ACCOUNT_ID"
	printf 'Management account: %s\n' "$MANAGEMENT_ACCOUNT_ID"

	if [ "$AVOS_ACCOUNT_ID" = "$MANAGEMENT_ACCOUNT_ID" ]; then
	  echo "RESULT: this is the Organizations management account"
	else
	  echo "RESULT: this is an Organizations member account"
	fi

List all SCPs

	  --filter SERVICE_CONTROL_POLICY \
	  --query 'Policies[].{

		Id:Id,

		Description:Description
	  }' \
	  --output table
